static inline long
fpr_floor(fpr x)
{
	uint64_t t;
	int64_t xi;
	int e, cc;

	e = (int)(x >> 52) & 0x7FF;
	t = x >> 63;
	xi = (int64_t)(((x << 10) | ((uint64_t)1 << 62))
		& (((uint64_t)1 << 63) - 1));
	xi = (xi ^ -(int64_t)t) + (int64_t)t;
	cc = 1085 - e;
	xi = fpr_irsh(xi, cc & 63);
	xi ^= (xi ^ -(int64_t)t) & -(int64_t)((uint32_t)(63 - cc) >> 31);
	return (long)xi;
}
