7684: /*
7685:  * ====================================================================
7686:  * FT1536 KEYGEN LEAF CERTIFICATE — MAIN ENGINE
7687:  * ====================================================================
7688:  */
7689: static int
7690: ft_keygen_leaf_certificate(fpr *tmp,
7691: 	const int16_t *f, const int16_t *g,
7692: 	const int16_t *F, const int16_t *G,
7693: 	unsigned logn, unsigned ter)
7694: {
7695: 	size_t n, hn, u;
7696: 	uint32_t bad;
7697: 	fpr q_squared;
7698: 	n = MKN(logn, ter);
7699: 	hn = n >> 1;
7700: 	if (logn != 10 || n != 1536 || hn != 768 || ter != 1) {
7701: 		return 0;
7702: 	}
7703: 	bad = 0;
7704: 	{
7705: 		fpr *tf, *tg, *tF, *tG;
7706: 		fpr *g00, *g10, *g11, *gxx;
7707: 		fpr *tree, *t3, *leaves, *scratch;
7708: 		size_t treesize;
7709: 		tf = tmp;
7710: 		tg = tf + n;
7711: 		tF = tg + n;
7712: 		tG = tF + n;
7713: 		g00 = tG + n;
7714: 		g10 = g00 + n;
7715: 		g11 = g10 + n;
7716: 		gxx = g11 + n;
7717: 		smallints_to_fpr_keygen(tf, f, logn, 1);
7718: 		smallints_to_fpr_keygen(tg, g, logn, 1);
7719: 		smallints_to_fpr_keygen(tF, F, logn, 1);
7720: 		smallints_to_fpr_keygen(tG, G, logn, 1);
7721: 		falcon_FFT3(tf, logn, 1);
7722: 		falcon_FFT3(tg, logn, 1);
7723: 		falcon_FFT3(tF, logn, 1);
7724: 		falcon_FFT3(tG, logn, 1);
7725: 		falcon_poly_neg_fft3(tf, logn, 1);
7726: 		falcon_poly_neg_fft3(tF, logn, 1);
7727: 		memcpy(g00, tg, n * sizeof *tg);
7728: 		falcon_poly_mulselfadj_fft3(g00, logn, 1);
7729: 		memcpy(gxx, tf, n * sizeof *tf);
7730: 		falcon_poly_mulselfadj_fft3(gxx, logn, 1);
7731: 		falcon_poly_add_fft3(g00, gxx, logn, 1);
7732: 		memcpy(g10, tG, n * sizeof *tG);
7733: 		falcon_poly_muladj_fft3(g10, tg, logn, 1);
7734: 		memcpy(gxx, tF, n * sizeof *tF);
7735: 		falcon_poly_muladj_fft3(gxx, tf, logn, 1);
7736: 		falcon_poly_add_fft3(g10, gxx, logn, 1);
7737: 		memcpy(g11, tG, n * sizeof *tG);
7738: 		falcon_poly_mulselfadj_fft3(g11, logn, 1);
7739: 		memcpy(gxx, tF, n * sizeof *tF);
7740: 		falcon_poly_mulselfadj_fft3(gxx, logn, 1);
7741: 		falcon_poly_add_fft3(g11, gxx, logn, 1);
7742: 		tree = gxx + n;
7743: 		treesize = (size_t)12 * n;
7744: 		t3 = tree + treesize;
7745: 		ffLDL_fft3_keygen(tree, g00, g10, g11, logn, t3);
7746: 		for (u = 0; u < hn; u ++) {
7747: 			uint32_t valid;
7748: 			uint64_t mask, xb;
7749: 			valid = (uint32_t)ft_fpr_is_positive_finite_keygen(g00[u]);
7750: 			valid &= (uint32_t)(1 ^ fpr_lt(g00[u], fpr_onehalf));
7751: 			bad |= valid ^ 1U;
7752: 			mask = (uint64_t)0 - (uint64_t)valid;
7753: 			xb = ft_fpr_bits_keygen(g00[u]);
7754: 			g00[u] = ft_fpr_from_bits_keygen(
7755: 				(xb & mask) | (ft_fpr_bits_keygen(fpr_one) & ~mask));
7756: 		}
7757: 		leaves = t3;
7758: 		scratch = leaves + n;
7759: 		ft_stable_top_branch_keygen(g00, leaves, scratch, &bad);
7760: 		q_squared = fpr_of(FT1536_KEYGEN_Q_SQUARED);
7761: 		for (u = 0; u < hn; u ++) {
7762: 			leaves[n - 1 - u] = ft_stable_positive_keygen(
7763: 				fpr_div(q_squared, leaves[u]), &bad);
7764: 		}
7765: 		for (u = 0; u < n; u ++) {
7766: 			uint64_t bits;
7767: 			uint32_t valid;
7768: 			leaves[u] = ft_stable_positive_keygen(leaves[u], &bad);
7769: 			bits = ft_fpr_bits_keygen(leaves[u]);
7770: 			valid = (uint32_t)(1 ^ (uint32_t)((bits
7771: 				- FT1536_LEAF_MIN_BITS) >> 63));
7772: 			valid &= (uint32_t)(1 ^ (uint32_t)((FT1536_LEAF_MAX_BITS
7773: 				- bits) >> 63));
7774: 			bad |= valid ^ 1U;
7775: 		}
7776: 		return bad == 0;
7777: 	}
